def sha(s):
    return hashlib.sha256(s.encode("utf-8")).hexdigest()


def stop(msg):
    sys.exit("ABORT, nothing written: " + msg)


def main():
    ids = {e[0] for e in EDITS} | {"W-1"}
    if APPROVED - ids:
        stop("unknown row ids " + str(sorted(APPROVED - ids)))
    if "W-1" in APPROVED and WIND_PLACEMENT not in ("A", "B"):
        stop("W-1 is approved but WIND_PLACEMENT is not 'A' or 'B'")
    if not DRY_RUN and not 5 <= datetime.datetime.now().hour < 15:
        stop("outside the 05:00-14:59 window")
    lines = {}
    orig = {}
    for k, p in FILES.items():
        with open(p, encoding="utf-8", newline="") as f:
            t = f.read()
        if "\r" in t or not t.endswith("\n"):
            stop(k + ": unexpected line endings")
        orig[k] = t
        lines[k] = t[:-1].split("\n")
        if len(lines[k]) != LINE_COUNT[k]:
            stop(k + ": line count " + str(len(lines[k])) + ", recorded " + str(LINE_COUNT[k]))
    todo = [e for e in EDITS if e[0] in APPROVED]
    if "W-1" in APPROVED and WIND_PLACEMENT == "A":
        todo.append(WIND_A)
    used = set()
    for rid, k, n, old, new, h in todo:
        line = lines[k][n - 1]
        whole = "\n".join(lines[k])
        if (k, n) in used:
            stop(rid + ": a second edit on " + k + " line " + str(n))
        used.add((k, n))
        if h is not None and sha(line) != h:
            stop(rid + ": " + k + " line " + str(n) + " differs from the recorded line (hash)")
        if line.count(old) != 1 or whole.count(old) != 1:
            stop(rid + ": the recorded current text is not found exactly once at " + k + " line " + str(n))
        if new in line or "\n" in new:
            stop(rid + ": already applied, or the new text would add a line")
    if "W-1" in APPROVED and WIND_PLACEMENT == "B" and WIND_B_BLOCK.strip() in "\n".join(lines["LC"]):
        stop("W-1 (B): the block is already present")
    for rid, k, n, old, new, h in todo:
        lines[k][n - 1] = lines[k][n - 1].replace(old, new, 1)
    out = {k: "\n".join(v) + "\n" for k, v in lines.items()}
    if "W-1" in APPROVED and WIND_PLACEMENT == "B":
        out["LC"] += WIND_B_BLOCK
    for k in out:
        extra = 2 if (k == "LC" and "W-1" in APPROVED and WIND_PLACEMENT == "B") else 0
        if out[k].count("\n") != LINE_COUNT[k] + extra:
            stop(k + ": the line count would change")
    done = sorted({e[0] for e in todo} | ({"W-1"} if "W-1" in APPROVED else set()))
    if DRY_RUN:
        print("DRY RUN: every check passed for", done, "- nothing written.")
        return
    for k, p in FILES.items():
        with open(p, encoding="utf-8", newline="") as f:
            if f.read() != orig[k]:
                stop(k + ": the file changed while the checks ran")
    for k, p in FILES.items():
        if out[k] != orig[k]:
            with open(p, "w", encoding="utf-8", newline="") as f:
                f.write(out[k])
    print("Applied:", done)


if __name__ == "__main__":
    main()
